/* PRODUCT FILTER */

function filterPlants(category, button) {

    document.querySelectorAll('.filter-btn').forEach(btn => {

        btn.classList.remove('active', 'btn-success');

        btn.classList.add('btn-outline-success');

    });

    button.classList.add('active', 'btn-success');

    button.classList.remove('btn-outline-success');


    document.querySelectorAll('.plant-item').forEach(item => {

        const matches =
        category === 'all' ||
        item.dataset.category === category;

        item.classList.toggle('d-none', !matches);

    });

}


/* SHOPPING CART */

const cart = [];


function addToCart(name, price) {

    const existing = cart.find(item => item.name === name);

    if (existing) {

        existing.quantity += 1;

    } else {

        cart.push({
            name: name,
            price: price,
            quantity: 1
        });

    }

    updateCart();

}


function updateCart() {

    const container = document.getElementById('cart-items');

    const total = cart.reduce(
        (sum, item) => sum + item.price * item.quantity,
        0
    );


    if (!cart.length) {

        container.innerHTML = `
            <p class="empty-cart text-center py-4 mb-0">
                <i class="bi bi-bag fs-2 d-block mb-2"></i>
                Your cart is empty.
            </p>
        `;

    } else {

        container.innerHTML = cart.map((item, index) => `

            <div class="d-flex justify-content-between align-items-center gap-3 py-2 border-bottom">

                <div>

                    <div class="fw-semibold">
                        ${item.name}
                    </div>

                    <small class="text-muted">
                        Rs. ${item.price.toLocaleString()} × ${item.quantity}
                    </small>

                </div>

                <div class="text-end">

                    <div class="fw-semibold">
                        Rs. ${(item.price * item.quantity).toLocaleString()}
                    </div>

                    <button
                    class="btn btn-sm btn-link text-danger p-0"
                    onclick="removeFromCart(${index})">

                        Remove

                    </button>

                </div>

            </div>

        `).join('');

    }


    document.getElementById('cart-total').textContent =
    'Rs. ' + total.toLocaleString();


    document.getElementById('cart-count').textContent =
    cart.reduce((sum, item) => sum + item.quantity, 0);

}


/* REMOVE PRODUCT */

function removeFromCart(index) {

    cart.splice(index, 1);

    updateCart();

}


/* SHOW CART */

function showCart() {

    updateCart();

    bootstrap.Modal.getOrCreateInstance(
        document.getElementById('cartModal')
    ).show();

}

const revealObserver = new IntersectionObserver(entries => {

    entries.forEach(entry => {

        if (entry.isIntersecting) {

            entry.target.classList.add('visible');

            revealObserver.unobserve(entry.target);

        }

    });

}, { threshold: 0.12 });


document.querySelectorAll('.reveal').forEach(element => {

    revealObserver.observe(element);

});
